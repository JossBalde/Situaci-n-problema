#include "video.h"
#include "episodio.h"
#include "pelicula.h"
#include <iostream>
#include <fstream>
using namespace std;
#define SERIES_FILE "series.csv"
#define PELICULAS_FILE "movies.csv"


int main() {
    Pelicula* peliculas = nullptr;
    Episodio* series = nullptr;
    int seriesSize = 0;
    int peliculasSize = 0;
    int catalogoSize = 0;
    Video** catalogo = nullptr;

    seriesSize = countDataLinesInCSV(SERIES_FILE);
    peliculasSize = countDataLinesInCSV(PELICULAS_FILE);



    int opcion;
        do{
        cout << "Menu:\n"
             << "1. Cargar archivo de datos\n"
             << "2. Mostrar los datos\n"
             << "3. Mostrar los videos en general con una cierta calificacion o de un cierto genero\n"
             << "4. Mostrar los episodios de una determinada serie con una calificacion determinada\n"
             << "5. Mostrar las peliculas con cierta calificacion\n"
             << "6. Calificar un video\n"
             << "0. Salir\n"
             << "Elige una opcion: ";
            cin >> opcion;
            switch (opcion) {
            case 1:
                if (seriesSize == -1) {
                    cerr << "No se pudo cargar el data set desde " << SERIES_FILE << "\n";
                    return 0;
                }
                series = new(nothrow) Episodio[seriesSize];
                if (series == nullptr) {
                    cerr << "No hubo memoria para el arreglo creado con datos de " << SERIES_FILE << "\n";
                    return 0;
                }
                if (!loadSerieFromCSV(SERIES_FILE, series, seriesSize)) {
                    cerr << "Error al cargar el data set de " << SERIES_FILE << "\n";
                    delete[] series;
                    return 0;
                }
                
                if (peliculasSize == -1) {
                    cerr << "No se pudo cargar el data set desde " << PELICULAS_FILE << "\n";
                    return 0;
                }
                peliculas = new(nothrow) Pelicula[peliculasSize];
                if (peliculas == nullptr) {
                    cerr << "No hubo memoria para el arreglo creado con datos de " << PELICULAS_FILE << "\n";
                    return 0;
                }
                if (!loadPeliculaFromCSV(PELICULAS_FILE, peliculas, peliculasSize)) {
                    cerr << "Error al cargar el data set de " << PELICULAS_FILE << "\n";
                    delete[] peliculas;
                    return 0;
                }
                catalogoSize = peliculasSize + seriesSize;

                catalogo = new(nothrow) Video * [catalogoSize];
                if (catalogo == nullptr) {
                    cerr << "No hubo memoria para el arreglo creado con datos de " << SERIES_FILE << " y " << PELICULAS_FILE << "\n";
                    return 0;
                }
                for (int i = 0; i < catalogoSize; i++)
                    catalogo[i] = nullptr;
                for (int i = 0; i < catalogoSize; i++) {
                    if (i < peliculasSize)
                        catalogo[i] = &peliculas[i];
                    else
                        catalogo[i] = &series[i - peliculasSize];
                }
                break;

            case 2:
                for (int i = 0; i < catalogoSize; i++)
                    catalogo[i]->mostrarDatos();
                break;
            case 3:
                int opcion2;
                std::cout << "¿Como deseas filtrar? Introduce 1 para filtrar por calificacion, o introduce 2 para filtrar por genero: ";
                std::cin >> opcion2;
                while (opcion2 != 1 && opcion2 != 2) {
                    std::cout << "Opcion invalida, por favor introduce 1 para calificacion o 2 para genero: ";
                    std::cin >> opcion2;
                }
                if (opcion2 == 1) {
                    int calificacionFiltro;
                    std::cout << "Dame la calificacion por la que quieres filtrar: ";
                    std::cin >> calificacionFiltro;
                    while (calificacionFiltro < 0 || calificacionFiltro > 5) {
                        std::cout << "Por favor introduce una calificacion valida (de 1 a 5): ";
                        std::cin >> calificacionFiltro;
                    }
                    for (int i = 0; i < catalogoSize; i++) {
                        if (catalogo[i]->getCalificacion() == calificacionFiltro) {
                            catalogo[i]->mostrarDatos();
                        }
                    }
                }
                else {
                    std::string generoFiltro;
                    int videosEncontrados = 0;
                    std::cout << "Dame el genero por el que quieres filtrar: ";
                    std::cin >> generoFiltro;
                    for (int i = 0; i < catalogoSize; i++) {
                        if (catalogo[i]->getGenero() == generoFiltro) {
                            catalogo[i]->mostrarDatos();
                            videosEncontrados++;
                        }
                    }
                    if (videosEncontrados == 0) {
                        std::cout << "No se encontro ningun titulo de ese genero \n";
                    }
                }
                break;
            case 4: {
                std::string nombreSerie;
                int calificacionEpisodio;
                int episodiosEncontrados = 0;
                std::cin.ignore();
                std::cout << "Dame el nombre de la serie: ";
                getline(std::cin, nombreSerie);
                std::cout << "Dame la calificacion por la que quieres filtrar: ";
                std::cin >> calificacionEpisodio;
                for (int i = 0; i < seriesSize; i++) {
                    if (series[i].getNombre() == nombreSerie && series[i].getCalificacion() == calificacionEpisodio) {
                        series[i].mostrarDatos();
                        episodiosEncontrados++;
                    }
                }
                if (episodiosEncontrados == 0) {
                    std::cout << "No se encontro ningun episodio con esas caracteristicas \n";
                }
                break;
            }
            case 5:
                int calificacion;
                std::cout << "Dame la calificacion por la que quieres filtrar las peliculas: ";
                std::cin >> calificacion;
                while (calificacion < 0 || calificacion > 5) {
                    std::cout << "Valor invalido, por favor introduce una calificacion valida (de 0 a 5): ";
                    std::cin >> calificacion;
                }
                for (int i = 0; i < peliculasSize; i++) {
                    if (peliculas[i].getCalificacion() == calificacion) {
                        peliculas[i].mostrarDatos();
                    }
                }
                break;
            case 6: {
                std::string idCalificar;
                float calificacionNueva;
                int videoEncontrado = 0;
                std::cout << "Dame la id del video a calificar: ";
                std::cin >> idCalificar;
                std::cout << "Dame la calificacion del video: ";
                std::cin >> calificacionNueva;
                for (int i = 0; i < catalogoSize; i++) {
                    if (catalogo[i]->getId() == idCalificar) {
                        catalogo[i]->setCalificacion(calificacionNueva);
                        videoEncontrado++;
                    }
                }
                if (videoEncontrado == 0) {
                    std::cout << "No existe el video con ID " << idCalificar;
                }
                break;
            }
            case 0:
                std::cout << "Gracias por usar este programa :) \n";
                break;
            }
        } while (opcion != 0);
    
        

    delete[] peliculas;
    delete[] series;
    delete[] catalogo;
    return 0;
}
